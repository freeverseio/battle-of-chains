import React from "react";
import BlockchainQuadrant from "./BlockchainQuadrant";
import PlayerBase from "./PlayerBase";
import { Blockchain, Player } from "./types";
interface MapContentProps {
  zoom: number;
  blockchains: Blockchain[];
  players: Player[];
  loggedInPlayer: string;
  openTooltip: string | null;
  toggleTooltip: (tooltipId: string) => void;
}

const MapContent: React.FC<MapContentProps> = ({
  zoom,
  blockchains,
  players,
  loggedInPlayer,
  openTooltip,
  toggleTooltip,
}) => {
  return (
    <div
      className="w-full h-full bg-cover bg-center relative"
      style={{
        backgroundImage: "url('/grid.jpg')",
        transform: `scale(${zoom})`,
        transition: "transform 0.3s ease-out",
      }}
    >
      {blockchains.map((blockchain) => (
        <BlockchainQuadrant key={blockchain.name} blockchain={blockchain} />
      ))}

      {players.map((player) =>
        player.bases.map((base) => (
          <PlayerBase
            key={`${player.name}-${base.chain}-${base.type}`}
            player={player}
            base={base}
            isLoggedInPlayer={player.name === loggedInPlayer}
            openTooltip={openTooltip}
            toggleTooltip={toggleTooltip}
          />
        ))
      )}
    </div>
  );
};

export default MapContent;
